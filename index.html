function copyCode(){
  if(!generatedCode)return;
  navigator.clipboard.writeText(generatedCode).then(()=>alert('代码已复制到剪贴板'));
}
function downloadHTML(){
  if(!generatedCode)return;
  const blob=new Blob([generatedCode],{type:'text/html'});
  const a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download='xi-design.html';a.click();
}
function openNew(){
  if(!generatedCode)return;
  const w=window.open('','_blank');
  w.document.write(generatedCode);
  w.document.close();
}
